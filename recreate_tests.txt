# CLI batch script compatible with typed rotation/time/deadbandSweep fields.
configure terminal

test set default

test delete "Rotation only (internal)"
test create "Rotation only (internal)"
type composite
device add "SPARKMAX/NEO 25"
duty 0.2
rotation limit 15.0
rotation encoderKey internal
rotation encoderMotorIndex 0
exit

test delete "Deadband sweep (internal)"
test create "Deadband sweep (internal)"
type deadbandSweep
device add "SPARKMAX/NEO 25"
deadbandSweep startDuty 0.0
deadbandSweep maxDuty 0.2
deadbandSweep stepDuty 0.01
deadbandSweep stepHoldSec 0.25
deadbandSweep motionThresholdRot 0.02
deadbandSweep requiredSamples 3
deadbandSweep encoderKey internal
deadbandSweep encoderMotorIndex 0
exit

test delete "Rotation + Time"
test create "Rotation + Time"
type composite
device add "FALCON 9"
duty 0.2
rotation limit 10.0
rotation encoderKey internal
rotation encoderMotorIndex 0
time timeout 2.0
time onTimeout fail
exit

test create "Time only"

time timeout 1.5

exit
