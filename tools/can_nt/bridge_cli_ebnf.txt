(* Bridge CLI Grammar (EBNF) *)

line           = ws? command ws? [ "?" ] ;
command        = common
               | exec
               | config
               | group
               | device
               | test ;

common         = "exit"
               | "end"
               | "help" [ ws name { ws name } ]
               | "ping"
               | "echo" ws ("on" | "off")
               | "sleep" ws number
               | "messages" ws ("beginner" | "medium" | "expert")
               | "tiu" ws ("on" | "off")
               | "debug" ws "grammar" [ ws "--json" ] [ ws "--dot" ws path ]
               | "debug" ws "grammar" [ ws "--dot" ws path ] [ ws "--json" ]
               | "quit" ;

exec           = show_exec
               | "configure" ws "terminal"
               | "connect"
               | "disconnect"
               | "runtime" ws ("activate" [ ws name ] | "deactivate")
               | "instantiate" ws "next" ws "motor"
               | "instantiate" ws "all" ws "devices"
               | "clear" ws ("stop-latch" | "safety-latch")
               | "reset" ws "zero-config" [ ws ("--yes" | "--clear-memory") ] [ ws ("--yes" | "--clear-memory") ]
               | "active" ws ("add" | "next" | ("show" [ ws "--json" ]))
               | "run" ws "test" [ ws name ]
               | "tests" ws ("select" ws name | "toggle" | tests_run_args | tests_wait_args)
               | "profile" ws name
               | "profile" ws "default" ws name
               | "profile" ws "create" ws name
               | "profile" ws "delete" ws name
               | "profile" ws "export" ws name ws path [ ws "--install-robot" ]
               | "profile" ws "device" ws "delete" ws name
               | "profile" ws "device" ws "show-all" ws name
               | "profiles" ws "init"
               | "profiles" ws "reload"
               | "profiles" ws "activate" ws name
               | "profiles" ws "export" ws path
               | "profiles" ws "push" ws path [ ws "--activate" ws name ]
               | "bindings" [ ws bindings_args ]
               | diagnose ;

show_exec      = "show" ws show_target [ ws show_flags ] ;

show_target    = "status"
               | "groups"
               | "group" ws name
               | "devices"
               | "device" ws name
               | "signals"
               | "signal" ws name
               | "commands"
               | "help"
               | "controllers"
               | "version"
               | "sources"
               | "controllers"
               | "version"
               | "sources"
               | "bindings"
               | "active"
               | "instantiated"
               | "faults"
               | "can-mappings" [ ws ("manufacturers" | "device-types" | "device-type") ]
               | "selected-device"
               | "safety-latch"
               | "runtime-state"
               | "config"
               | "config" ws "local-raw"
               | "config" ws "dirty"
               | "profiles"
               | "profile" [ ws name ]
               | "tests" [ ws "global" | ws "sets" [ ws name ] ]
               | "test" ws "sets" [ ws name ]
               | "test" ws name [ ws "normalized" ]
               | "topology"
               | "topology" ws "neighbors"
               | "topology" ws "nodes"
               | "topology" ws "edges"
               | "topology" ws "node" ws name
               | "neighbors" ws name
               | "device-group" ws name
               | "device-usage" ws name
               | "message-level"
               | "workspace" ;

show_source    = "robot" | "local" | "both" ;
show_flags     = show_flag { ws show_flag } ;
show_flag      = show_source | "--json" | "--pretty" | "--all" | "--grouped" ;

config         = "group" ws name
               | "group" ws "create" ws name
               | "group" ws "delete" ws name
               | "group" ws "rename" ws name ws name
               | "group" ws "clear" [ ws name ]
               | "no" ws "group" ws name
               | "no" ws "device" ws name
               | "group" ws "member" ws ("assign" ws name ws name
                                         | "remove" ws name ws name
                                         | "enable" ws name ws name
                                         | "disable" ws name ws name
                                         | "toggle" ws name ws name
                                         | "assign" ws ("all" | "next") ws name)
               | "member" ws ("assign" ws name
                               | "remove" ws name
                               | "enable" ws name
                               | "disable" ws name
                               | "toggle" ws name
                               | "assign" ws ("all" | "next"))
               | "instantiate" ws "next" ws "motor"
               | "instantiate" ws "all" ws "devices"
               | "clear" ws ("stop-latch" | "safety-latch")
               | "copy" ws "group" ws name ws name
               | "profile" ws name
               | "profile" ws "default" ws name
               | "profile" ws "create" ws name
               | "profile" ws "delete" ws name
               | "profile" ws "export" ws name ws path [ ws "--install-robot" ]
               | "profile" ws "device" ws "delete" ws name
               | "profile" ws "device" ws "show-all" ws name
               | "selected-device" ws name
               | "selected-mode" ws ("on" | "off")
               | "merge" ws "config" ws path
               | "import" ws "config" ws path
               | "export" ws "runtime-groups" ws path
               | "export" ws "cli-script" ws path
               | "save" ws "tests" ws path [ ws "--force" ]
               | "save" ws "config" ws path [ ws "--force" ]
               | "save" ws "bridge-config" ws path [ ws "--force" ]
               | "save" ws "runtime-groups" ws path [ ws "--force" ]
               | "save" ws "profiles" [ ws path ] [ ws "--force" ]
               
               | "save" ws "all" [ ws "--prompt" ] [ ws "--force" ]
               | "save" ws "sources" [ ws "--force" ]
               | "load" ws "sources"
               | "rename" ws "device" ws name ws name
               | "rename" ws name ws name
               | "device" ws name
               | "device" ws name ws "set" ws device_field ws value_or_text
               | ("validate" | "val") ws "all" [ ws "--verbose" ]
               | "validate" ws "config" [ ws path ] [ ws "--all" ] [ ws "--verbose" ]
               | ("validate" | "val") ws "file" ws path [ ws "--repair" ] [ ws "--verbose" ]
               | ("validate" | "val") ws "profiles" [ ws ("robot" | "local") ] [ ws "--active" ] [ ws "--verbose" ]
               | ("validate" | "val") ws "tests" [ ws "--active-set" ] [ ws "--verbose" ]
               | ("validate" | "val") ws "topology" [ ws "--verbose" ]
               | ("validate" | "val") ws "bindings" [ ws path ] [ ws "--verbose" ]
               | ("validate" | "val") ws "can-mappings" [ ws path ] [ ws "--verbose" ]
               | ("validate" | "val") ws "script" ws path [ ws "--verbose" ]
               | "recover" ws ("list" | "last-good" | "from" ws name)
               | "profiles" ws "init"
               | "profiles" ws "reload"
               | "profiles" ws "activate" ws name
               | "runtime" ws ("activate" [ ws name ] | "deactivate")
               | "profiles" ws "export" ws path
               | "profiles" ws "push" ws path [ ws "--activate" ws name ]
               | "config" ws "push" ws path [ ws "--activate" ws name ]
               | "run" ws "test" [ ws name ]
               | "topology" ws "neighbor-ports" ws "set" ws name ws name ws name ws name
               | "topology" ws "neighbor-ports" ws "delete" ws name ws name
               | "topology" ws "neighbor-ports" ws "clear" ws name
               | "topology" ws "neighbor-auto" ws "all" [ ws name ]
               | "topology" ws "neighbor-auto" ws "node" ws name
               | ("show" | "ls") ws show_target [ ws show_flags ]
               | diagnose
               | "bindings" [ ws bindings_args ]
               | "can-mappings" [ ws mappings_args ]
               | "tests" [ ws tests_args ]
               | "test" ws ("import" ws name ws path [ ws "set" ws name ]
                           | "export" ws name ws path
                           | "validate" [ ws name ] [ ws "--json" ] [ ws "--pretty" ]
                           | "validate" [ ws "--json" ] [ ws "--pretty" ] [ ws name ]
                           | "delete" ws name
                           | "cleanup" ws "stale"
                           | "set" ws ("create" ws name
                                      | "delete" ws name
                                      | "add" ws name ws name
                                      | "remove" ws name ws name
                                      | "default" ws name)) ;

diagnose       = "diagnose" ws name ws name ;

bindings_args  = "show" [ ws ("controllers" | "bindings") ] [ ws show_flags ]
               | "clear"
                | "controller" ws ("add" ws name ws name ws number
                                    | "set" ws name ws field ws value_text
                                    | "rename" ws name ws name
                                    | "list"
                                    | "no" ws name)
               | "no" ws "controller" ws name
               | "binding" ws ("add" ws name ws name ws name ws name ws name [ ws "invert" ws ("on" | "off") ws "deadband" ws number ]
                                | "set" ws number ws field ws value_text
                                | "delete" ws number)
               | "load" ws path
               | "save" ws path
               | "validate" [ ws path ] ;

mappings_args  = "show" [ ws ("manufacturers" | "device-types" | "device-type") ] [ ws show_flags ]
               | "clear"
               | "manufacturer" ws ("set" ws number ws value_text | "delete" ws number | "no" ws number)
               | "device-type" ws ("set" ws number ws value_text | "delete" ws number | "no" ws number)
               | "load" ws path
               | "save" ws path
               | "validate" [ ws path ] ;

tests_args     = "templates"
               | "clear"
               | "select" ws name
               | "toggle"
               | tests_run_args
               | tests_wait_args ;

tests_run_args = "run" [ ws "--wait" ] [ ws "--timeout" ws number ]
               | "run" [ ws "--timeout" ws number ] [ ws "--wait" ]
               | "run-all" [ ws "--wait" ] [ ws "--timeout" ws number ]
               | "run-all" [ ws "--timeout" ws number ] [ ws "--wait" ] ;

tests_wait_args = "wait"
                | "wait" ws "--run" ws number
                | "wait" ws "--timeout" ws number
                | "wait" ws "--run" ws number ws "--timeout" ws number
                | "wait" ws "--timeout" ws number ws "--run" ws number ;

group          = "show"
               | "show" ws "members"
               | "show" ws "binding"
               | "show" ws show_target [ ws show_flags ]
               | "member" ws ("assign" ws name
                               | "remove" ws name
                               | "enable" ws name
                               | "disable" ws name
                               | "toggle" ws name
                               | "assign" ws ("all" | "next"))
               | "clear"
               | "bind" ws "list"
               | "bind" ws "explain" ws input
               | "bind" ws "test" ws input
               | "bind" ws input ws "analog"
               | "bind" ws input ws ("hold" | "toggle" | "jog-forward" | "jog-reverse") ws value
               | "no" ws "bind"
               | "enable"
               | "disable"
               | "run" ws "test" [ ws name ] ;

device         = "show"
               | "show" ws show_target [ ws show_flags ]
               | "set" ws device_field ws value_or_text
               | device_field ws value_or_text
               | "no" ws device_field
               | "delete" ;

test           = "show"
               | "type" ws ("joystick"
                            | "button"
                            | "composite"
                            | "deadbandsweep" | "deadbandSweep"
                            | "deviceaction" | "deviceAction")
               | "device" ws "add" ws name [ ws "role" ws ("primary" | "observer") ]
               | "device" ws "create" ws name ws "type" ws "TestTimer"
               | "no" ws "device" ws name
               | "command" ws name ws "=" ws value_or_text
               | ("until" | "expect" | "success" | "abort") ws name ws ("==" | "!=" | ">=" | ">" | "<=" | "<") ws value_or_text
               | "passive" ws ("true" | "false" | "on" | "off")
               | "manual_stop" ws ("true" | "false" | "on" | "off")
               | "inputsource" ws name
               | "deadband" ws number
               | "duty" ws number
               | "action" ws name
               | "color" ws name
               | "pattern" ws name
               | "brightness" ws number
               | "duration" ws number
               | "rotation" ws "limit" ws number
               | "rotation" ws ("encoderKey" | "encoderSource") ws name
               | "rotation" ws ("encoderMotorIndex" | "encoderCountsPerRev") ws number
               | "time" ws "timeout" ws number
               | "time" ws "onTimeout" ws name
               | "hold" ws "onRelease" ws name
               | "limitswitch" ws ("onHit" | "id") ws name
               | "deadbandsweep" ws ("startDuty" | "maxDuty" | "stepDuty" | "stepHoldSec" | "motionThresholdRot") ws number
               | "deadbandsweep" ws "requiredSamples" ws number
               | "deadbandsweep" ws ("encoderKey" | "encoderSource") ws name
               | "deadbandsweep" ws ("encoderMotorIndex" | "encoderCountsPerRev") ws number
               | "enabled" ws ("true" | "false" | "on" | "off")
               | "termination" ws "hold"
               | "termination" ws "time" ws number
               | "termination" ws "rotation" ws number
               | "termination" ws "limitswitch" [ ws name ] ;

(* Lexical conventions *)

name           = token ;
input          = token ;
value          = number ;
value_or_text  = number | value_text ;
path           = token ;
field          = token ;
device_field   = "interface"
               | "deviceInterface"
               | "manufacturer"
               | "deviceType"
               | "id"
               | "model"
               | "type"
               | "dio"
               | "invert"
               | "pwm"
               | "analog"
               | "attachments"
               | "terminator"
               | "vendor"
               | "role"
               | "notes"
               | "tags"
               | "limits" ;
value_text     = (token | "default") { ws (token | "default") } ;

token          = token_char { token_char } ;
token_char     = ? any non-whitespace character ? ;

number         = ["+"|"-"] digit { digit } [ "." digit { digit } ] ;
digit          = "0"..."9" ;
ws             = { " " | "\t" } ;
